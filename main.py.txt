import json

from core_uniquifier import ImageUniquifier, Settings


class App(tk.Tk):

        super().__init__()
        self.title("UniqLab — JPEG Uniquifier")

        self.minsize(900, 620)

    # ---------- UI ----------
    def _build_ui(self):
        root = ttk.Frame(self, padding=12)
        root.pack(fill="both", expand=True)

        # left controls / right console
        left = ttk.Frame(root)
        right = ttk.Frame(root)
        left.pack(side="left", fill="both", expand=True)
        right.pack(side="right", fill="both", expand=True, padx=(12, 0))

        # Title bar
        title = ttk.Label(left, text="UNIQLAB / Visual Uniquifier (JPG/JPEG)", font=("Segoe UI", 14, "bold"))
        title.pack(anchor="w", pady=(0, 8))

        # Scroll area for controls
        canvas = tk.Canvas(left, highlightthickness=0)
        scroll = ttk.Scrollbar(left, orient="vertical", command=canvas.yview)
        controls = ttk.Frame(canvas)

        controls.bind(
            "<Configure>",
            lambda e: canvas.configure(scrollregion=canvas.bbox("all"))
        )
        canvas.create_window((0, 0), window=controls, anchor="nw")
        canvas.configure(yscrollcommand=scroll.set)

        canvas.pack(side="left", fill="both", expand=True)
        scroll.pack(side="right", fill="y")

        # --- Sections ---
        self._section_io(controls)
        self._section_pipeline(controls)
        self._section_effects(controls)
        self._section_color(controls)
        self._section_save(controls)
        self._section_config(controls)

        ttk.Label(right, text="Console", font=("Segoe UI", 12, "bold")).pack(anchor="w")
        self.console = tk.Text(right, height=10, bg="#0b0f14", fg="#9ef01a", insertbackground="#9ef01a")
        self.console.pack(fill="both", expand=True)
        self._log("Ready. Put JPG/JPEG into ./input and click START.\n")

        self.pbar.pack(fill="x", pady=(10, 6))
        self.pbar_label = ttk.Label(right, text="0 / 0")
        self.pbar_label.pack(anchor="w")

        # start button

        self.start_btn.pack(fill="x", pady=(10, 0))

    def _group(self, parent, title):

    def _log(self, s: str):

    # ---------- actions ----------
    def open_output(self):
        out = Path("output").resolve()
        out.mkdir(exist_ok=True)
        try:
            os = __import__("os")
            import sys
            if sys.platform.startswith("win"):
                os.startfile(str(out))
            elif sys.platform == "darwin":
                os.system(f'open "{out}"')
            else:
                os.system(f'xdg-open "{out}"')
        except Exception:
            messagebox.showinfo("Output", f"Output folder: {out}")

    def gather_settings(self) -> Settings:
        st = Settings()

        st.generations = int(self.var_generations.get())
        st.use_background = bool(self.var_use_bg.get())
        st.canvas_w = int(self.var_cw.get())
        st.canvas_h = int(self.var_ch.get())
        st.background_blur = float(self.var_bg_blur.get())
        st.random_position = bool(self.var_rand_pos.get())
        st.preserve_exif = bool(self.var_preserve_exif.get())

        st.rotate_enabled = bool(self.var_rot.get())
        st.rotate_max_deg = float(self.var_rot_max.get())

        st.resize_enabled = bool(self.var_resize.get())
        st.resize_min_scale = float(self.var_min_s.get())
        st.resize_max_scale = float(self.var_max_s.get())

        st.mirror_x = bool(self.var_mx.get())
        st.mirror_y = bool(self.var_my.get())

        st.blur_enabled = bool(self.var_blur.get())
        st.blur_max_radius = float(self.var_blur_max.get())

        st.noise_enabled = bool(self.var_noise.get())
        st.noise_intensity = float(self.var_noise_int.get())

        st.lines_enabled = bool(self.var_lines.get())
        st.lines_count = int(self.var_lines_n.get())

        st.snow_enabled = bool(self.var_snow.get())
        st.snow_density = float(self.var_snow_d.get())

        st.rounded_enabled = bool(self.var_round.get())
        st.rounded_radius = int(self.var_round_r.get())

        st.pixelate_enabled = bool(self.var_pix.get())
        st.pixel_size = int(self.var_pix_sz.get())

        st.color_enabled = bool(self.var_col.get())
        st.sat_min = float(self.var_sat_min.get()); st.sat_max = float(self.var_sat_max.get())
        st.bri_min = float(self.var_bri_min.get()); st.bri_max = float(self.var_bri_max.get())
        st.con_min = float(self.var_con_min.get()); st.con_max = float(self.var_con_max.get())
        st.sha_min = float(self.var_sha_min.get()); st.sha_max = float(self.var_sha_max.get())

        st.jpeg_quality_min = int(self.var_qmin.get())
        st.jpeg_quality_max = int(self.var_qmax.get())
        st.target_kb_min = int(self.var_tmin.get())
        st.target_kb_max = int(self.var_tmax.get())
        st.make_zip = bool(self.var_zip.get())

        # safety clamp
        if st.jpeg_quality_min > st.jpeg_quality_max:
            st.jpeg_quality_min, st.jpeg_quality_max = st.jpeg_quality_max, st.jpeg_quality_min
        if st.resize_min_scale > st.resize_max_scale:
            st.resize_min_scale, st.resize_max_scale = st.resize_max_scale, st.resize_min_scale

        return st

    def on_start(self):
        try:
            st = self.gather_settings()
        except Exception as e:
            messagebox.showerror("Settings error", str(e))
            return

        self.start_btn.config(state="disabled")
        self.console.delete("1.0", "end")
        self._log(">>> RUN\n")
        self._log(f"Settings: {json.dumps(st.to_dict(), ensure_ascii=False)}")

        def worker():
            try:
                def progress(done, total):
                    self.after(0, lambda: self._set_progress(done, total))

                def logline(s):
                    self.after(0, lambda: self._log(s))

                results, zip_path = self.u.process_all(st, progress_cb=progress, log_cb=logline)

                def finish():
                    self._log("\n>>> DONE")
                    self._log(f"Saved: {len(results)} files in ./output")
                    if zip_path:
                        self._log(f"ZIP: {zip_path}")
                    self.start_btn.config(state="normal")

                self.after(0, finish)

            except Exception as e:
                def err():
                    self._log(f"\n[ERROR] {e}")
                    messagebox.showerror("Error", str(e))
                    self.start_btn.config(state="normal")
                self.after(0, err)

        threading.Thread(target=worker, daemon=True).start()

    def _set_progress(self, done, total):
        self.pbar["maximum"] = max(1, total)
        self.pbar["value"] = done
        self.pbar_label.config(text=f"{done} / {total}")

    def copy_config(self):
        st = self.gather_settings()
        text = json.dumps(st.to_dict(), ensure_ascii=False, indent=2)
        self.clipboard_clear()
        self.clipboard_append(text)
        self._log("Config copied to clipboard.")

    def paste_config(self):
        try:
            text = self.clipboard_get()
            d = json.loads(text)
            st = Settings.from_dict(d)
        except Exception as e:
            messagebox.showerror("Paste error", f"Clipboard is not valid JSON config.\n{e}")
            return

        # apply back to UI
        self.var_generations.set(st.generations)
        self.var_use_bg.set(st.use_background)
        self.var_cw.set(st.canvas_w); self.var_ch.set(st.canvas_h)
        self.var_bg_blur.set(st.background_blur)
        self.var_rand_pos.set(st.random_position)
        self.var_preserve_exif.set(st.preserve_exif)

        self.var_rot.set(st.rotate_enabled); self.var_rot_max.set(st.rotate_max_deg)
        self.var_resize.set(st.resize_enabled)
        self.var_min_s.set(st.resize_min_scale); self.var_max_s.set(st.resize_max_scale)
        self.var_mx.set(st.mirror_x); self.var_my.set(st.mirror_y)
        self.var_blur.set(st.blur_enabled); self.var_blur_max.set(st.blur_max_radius)
        self.var_noise.set(st.noise_enabled); self.var_noise_int.set(st.noise_intensity)
        self.var_lines.set(st.lines_enabled); self.var_lines_n.set(st.lines_count)
        self.var_snow.set(st.snow_enabled); self.var_snow_d.set(st.snow_density)
        self.var_round.set(st.rounded_enabled); self.var_round_r.set(st.rounded_radius)
        self.var_pix.set(st.pixelate_enabled); self.var_pix_sz.set(st.pixel_size)

        self.var_col.set(st.color_enabled)
        self.var_sat_min.set(st.sat_min); self.var_sat_max.set(st.sat_max)
        self.var_bri_min.set(st.bri_min); self.var_bri_max.set(st.bri_max)
        self.var_con_min.set(st.con_min); self.var_con_max.set(st.con_max)
        self.var_sha_min.set(st.sha_min); self.var_sha_max.set(st.sha_max)

        self.var_qmin.set(st.jpeg_quality_min); self.var_qmax.set(st.jpeg_quality_max)
        self.var_tmin.set(st.target_kb_min); self.var_tmax.set(st.target_kb_max)
        self.var_zip.set(st.make_zip)

        self._log("Config pasted from clipboard.")
